async def check_subscription(user_id: int, bot) -> bool: # Проверяет, подписан ли пользователь на канал
    try:
        member = await bot.get_chat_member(chat_id=CHANNEL_ID, user_id=user_id)
        return member.status in ["member", "administrator", "creator"]
    except TelegramBadRequest:
        return False

@router.callback_query(F.data == 'register_player')
async def register(callback: CallbackQuery, state:FSMContext):
    await callback.answer('') # нужно отвечать чисто callback, иначе выбранная кнопка продолжить мигать
    await state.set_state(Reg_player.name)
    await callback.message.answer('Введите ваше имя 👇')

@router.message(Reg_player.name)
async def register_name(message: Message, state:FSMContext):
    user = message.from_user

    if user.username:
        user_link = f"https://t.me/{user.username}"
    else:
        user_link = f"tg://user?id={user.id}"
        
    await state.update_data(link = user_link)
    await state.update_data(name = message.text)
    await state.set_state(Reg_player.age)
    await message.answer('Введите ваш возраст 👇')
